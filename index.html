My Console-Based Game

This section contains the Java source code for my final Performance Task in CC102. The program was developed as a console-based game using Java programming concepts. 


import java.util.Scanner;
import java.util.Random;

public class Main {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        Random random = new Random();

        String[] easyWords = {
            "cat", "dog", "fish", "book", "apple"
        };

        String[] mediumWords = {
            "computer", "keyboard", "internet", "program"
        };

        String[] hardWords = {
            "algorithm",
            "encryption",
            "cybersecurity",
            "programming",
            "pneumonoultramicroscopicsilicavolcanoconiosis"
        };

        try {

            System.out.println("=== GUESS THE WORD ===");
            System.out.println("1. Easy");
            System.out.println("2. Medium");
            System.out.println("3. Hard");

            System.out.print("Choose difficulty: ");
            int difficulty = input.nextInt();
            input.nextLine();

            String[] words;
            int attempts;

            if (difficulty == 1) {
                words = easyWords;
                attempts = 7;
            }
            else if (difficulty == 2) {
                words = mediumWords;
                attempts = 6;
            }
            else if (difficulty == 3) {
                words = hardWords;
                attempts = 5;
            }
            else {
                System.out.println("Invalid choice!");
                return;
            }

            String secretWord =
                words[random.nextInt(words.length)];

            char[] guessedWord =
                new char[secretWord.length()];

            for (int i = 0; i < guessedWord.length; i++) {
                guessedWord[i] = '_';
            }

            boolean won = false;
            String guessedLetters = "";

            while (attempts > 0) {

                System.out.println(
                    "\nWord: " + new String(guessedWord)
                );

                System.out.println(
                    "Attempts left: " + attempts
                );

                System.out.print("Guess a letter: ");

                char guess = input.nextLine()
                    .toLowerCase()
                    .charAt(0);

                if (guessedLetters.indexOf(guess) != -1) {

                    System.out.println(
                        "You already guessed that letter!"
                    );

                    continue;
                }

                guessedLetters += guess;

                boolean found = false;

                for (int i = 0;
                     i < secretWord.length();
                     i++) {

                    if (secretWord.charAt(i) == guess) {

                        guessedWord[i] = guess;
                        found = true;
                    }
                }

                if (found) {
                    System.out.println(
                        "Correct letter!"
                    );
                }
                else {
                    System.out.println(
                        "Wrong letter!"
                    );

                    attempts--;
                }

                if (new String(guessedWord)
                        .equals(secretWord)) {

                    won = true;
                    break;
                }
            }

            if (won) {

                System.out.println("\nYOU WON!");

                System.out.println(
                    "The word was: " + secretWord
                );
            }
            else {

                System.out.println("\nGAME OVER!");

                System.out.println(
                    "The word was: " + secretWord
                );
            }

        }
        catch (Exception e) {

            System.out.println();
            System.out.println("Invalid");
        }

        input.close();
    }
}

                        name[i] + word
                    );

                    );

                    found = true;

            if (!found) {

                System.out.println(
                    sname + " Cannot Be Found!"
                );
            }

        }

}
